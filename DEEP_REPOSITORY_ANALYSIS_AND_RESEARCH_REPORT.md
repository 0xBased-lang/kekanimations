### **Step 2: Batch Process 43 Layers** (Priority 2)

**Script**: `scripts/batch_animate_layers.py` (NEW)

**Implementation**:
```python
import os
from pathlib import Path
from animate_layer import LayerAnimator

LAYER_TYPES = [
    "background", "body", "eyes", "hat",
    "glasses", "clothes", "tools", "special",
    "tattoo", "style"
]

def batch_animate_layers():
    layers_path = Path("media_files/nft_layers")
    output_path = Path("animated_layers")
    output_path.mkdir(exist_ok=True)

    for layer_type in LAYER_TYPES:
        layer_folder = layers_path / layer_type

        if not layer_folder.exists():
            continue

        print(f"\n🎬 Animating {layer_type.upper()}...")

        for layer_file in layer_folder.glob("*.png"):
            trait_name = layer_file.stem

            print(f"  • {trait_name}...")

            # Load layer
            layer_img = Image.open(layer_file)

            # Animate
            animator = LayerAnimator(layer_type, trait_name)
            frames = animator.animate(layer_img)

            # Save frame sequence
            output_folder = output_path / layer_type / trait_name
            output_folder.mkdir(parents=True, exist_ok=True)

            for i, frame in enumerate(frames):
                frame.save(
                    output_folder / f"frame_{i:02d}.png",
                    "PNG"
                )

            print(f"    ✅ Saved {len(frames)} frames")

if __name__ == "__main__":
    batch_animate_layers()
```

**Time**: 3-5 hours to process 43 layers (mostly automated)
